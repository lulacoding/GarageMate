# GarageMate — Plan 04: Map Screen

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Map screen showing nearby garage sales as custom map annotations, with a bottom sheet detail card on tap. Uses `react-native-maps` on iOS/Android and `react-leaflet` on Web.

**Architecture:** `Platform.OS === 'web'` check at render time to switch between `react-leaflet` and `react-native-maps`. Both use the same `useListings` hook from Plan 03 with a fixed 20km radius. Tapping an annotation sets `selectedListing` state which renders a bottom card. Listings are loaded once on mount — no dynamic viewport querying in v1.

**Tech Stack:** `react-native-maps`, `react-leaflet`, `expo-location`, `useListings` hook (Plan 03), TypeScript

**Skills:** `@mobile-developer` (platform-conditional imports), `@expo-best-practices` (location + maps), `@uncodixfy` (invoke before writing any JSX/styling — no generic AI aesthetics), `@code-quality` (TypeScript strictness)

```tsx
import MapView, { Marker, Region } from 'react-native-maps';
import { StyleSheet } from 'react-native';
import { Listing, LocationCoords } from '@/types';

interface Props {
  center: LocationCoords;
  listings: Listing[];
  onSelectListing: (listing: Listing) => void;
}

export function MapNative({ center, listings, onSelectListing }: Props) {
  const region: Region = {
    latitude: center.lat,
    longitude: center.lng,
    latitudeDelta: 0.3,
    longitudeDelta: 0.3,
  };

  return (
    <MapView style={styles.map} initialRegion={region} showsUserLocation>
      {listings.map((listing) => (
        <Marker
          key={listing.id}
          coordinate={{ latitude: listing.lat, longitude: listing.lng }}
          title={listing.title}
          onPress={() => onSelectListing(listing)}
        />
      ))}
    </MapView>
  );
}

const styles = StyleSheet.create({
  map: { flex: 1 },
});
```

### Task 2: Web map component

**Files:**
- Create: `components/MapWeb.tsx`

- [ ] **Step 1: Create components/MapWeb.tsx**

```tsx
import { useEffect } from 'react';
import { MapContainer, TileLayer, Marker, Popup, useMap } from 'react-leaflet';
import 'leaflet/dist/leaflet.css';
import L from 'leaflet';
import { Listing, LocationCoords } from '@/types';

// Fix default marker icon missing in webpack builds
delete (L.Icon.Default.prototype as any)._getIconUrl;
L.Icon.Default.mergeOptions({
  iconRetinaUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon-2x.png',
  iconUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png',
  shadowUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png',
});

interface Props {
  center: LocationCoords;
  listings: Listing[];
  onSelectListing: (listing: Listing) => void;
}

function RecenterMap({ center }: { center: LocationCoords }) {
  const map = useMap();
  useEffect(() => { map.setView([center.lat, center.lng]); }, [center.lat, center.lng]);
  return null;
}

export function MapWeb({ center, listings, onSelectListing }: Props) {
  return (
    <MapContainer
      center={[center.lat, center.lng]}
      zoom={13}
      style={{ height: '100%', width: '100%' }}
    >
      <TileLayer
        attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a>'
        url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
      />
      <RecenterMap center={center} />
      {listings.map((listing) => (
        <Marker
          key={listing.id}
          position={[listing.lat, listing.lng]}
          eventHandlers={{ click: () => onSelectListing(listing) }}
        >
          <Popup>{listing.title}</Popup>
        </Marker>
      ))}
    </MapContainer>
  );
}
```

```tsx
import { useEffect, useState } from 'react';
import { View, Text, TouchableOpacity, StyleSheet, Platform, ActivityIndicator } from 'react-native';
import * as Location from 'expo-location';
import { useRouter } from 'expo-router';
import { useListings } from '@/hooks/useListings';
import { Listing, LocationCoords } from '@/types';

// Platform-conditional map import
const MapComponent = Platform.OS === 'web'
  ? require('@/components/MapWeb').MapWeb
  : require('@/components/MapNative').MapNative;

export default function MapScreen() {
  const [center, setCenter] = useState<LocationCoords | null>(null);
  const [locationError, setLocationError] = useState(false);
  const [selectedListing, setSelectedListing] = useState<Listing | null>(null);
  const router = useRouter();

  const { listings, loading, error } = useListings(center, 20);

  useEffect(() => {
    (async () => {
      const { status } = await Location.requestForegroundPermissionsAsync();
      if (status !== 'granted') { setLocationError(true); return; }
      const loc = await Location.getCurrentPositionAsync({});
      setCenter({ lat: loc.coords.latitude, lng: loc.coords.longitude });
    })();
  }, []);

  if (!center && !locationError) {
    return (
      <View style={styles.center}>
        <ActivityIndicator size="large" />
        <Text style={styles.loadingText}>Getting your location...</Text>
      </View>
    );
  }

  if (locationError) {
    return (
      <View style={styles.center}>
        <Text style={styles.errorText}>Location permission required to show the map.</Text>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      {loading && (
        <View style={styles.loadingOverlay}>
          <ActivityIndicator size="small" color="#2563eb" />
        </View>
      )}
      <MapComponent
        center={center!}
        listings={listings}
        onSelectListing={setSelectedListing}
      />
      {selectedListing && (
        <View style={styles.bottomCard}>
          <Text style={styles.cardTitle} numberOfLines={1}>{selectedListing.title}</Text>
          <Text style={styles.cardAddress} numberOfLines={1}>{selectedListing.address}</Text>
          <Text style={styles.cardDate}>
            {selectedListing.startDate.toDate().toLocaleDateString('en-AU', { weekday: 'short', month: 'short', day: 'numeric' })}
            {' · '}{selectedListing.startTime}–{selectedListing.endTime}
          </Text>
          <View style={styles.cardButtons}>
            <TouchableOpacity
              style={styles.viewBtn}
              onPress={() => router.push(`/listing/${selectedListing.id}`)}
            >
              <Text style={styles.viewBtnText}>View Details</Text>
            </TouchableOpacity>
            <TouchableOpacity style={styles.closeBtn} onPress={() => setSelectedListing(null)}>
              <Text style={styles.closeBtnText}>✕</Text>
            </TouchableOpacity>
          </View>
        </View>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1 },
  center: { flex: 1, justifyContent: 'center', alignItems: 'center', padding: 24 },
  loadingText: { marginTop: 12, color: '#6b7280' },
  errorText: { color: '#dc2626', textAlign: 'center' },
  loadingOverlay: { position: 'absolute', top: 16, right: 16, zIndex: 10, backgroundColor: '#fff', borderRadius: 8, padding: 8, elevation: 4 },
  bottomCard: { position: 'absolute', bottom: 0, left: 0, right: 0, backgroundColor: '#fff', padding: 16, borderTopLeftRadius: 16, borderTopRightRadius: 16, elevation: 8, shadowColor: '#000', shadowOpacity: 0.12, shadowRadius: 12, shadowOffset: { width: 0, height: -4 } },
  cardTitle: { fontSize: 17, fontWeight: '600', marginBottom: 4 },
  cardAddress: { fontSize: 13, color: '#6b7280', marginBottom: 4 },
  cardDate: { fontSize: 13, color: '#9ca3af', marginBottom: 12 },
  cardButtons: { flexDirection: 'row', gap: 8 },
  viewBtn: { flex: 1, backgroundColor: '#2563eb', padding: 12, borderRadius: 8, alignItems: 'center' },
  viewBtnText: { color: '#fff', fontWeight: '600' },
  closeBtn: { padding: 12, borderWidth: 1, borderColor: '#e5e7eb', borderRadius: 8, alignItems: 'center', justifyContent: 'center' },
  closeBtnText: { color: '#6b7280' },
});
```
