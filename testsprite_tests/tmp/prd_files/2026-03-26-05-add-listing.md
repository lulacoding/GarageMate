# GarageMate — Plan 05: Add Listing Screen

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Add Listing screen with image picker (max 3), address autocomplete, date/time picker, full validation, Firebase Storage upload, and Firestore write. Also supports Edit mode (pre-filled from existing listing).

**Architecture:** Single `AddListingScreen` component handles both add and edit modes via optional `listingId` route param. Images are uploaded to Firebase Storage first, then the Firestore doc is written atomically. `geofire-common` computes the geohash from the selected address lat/lng. Edit mode preserves existing image URLs and only uploads new ones.

**Tech Stack:** `expo-image-picker`, Firebase Storage, Firestore, `react-native-google-places-autocomplete`, `geofire-common`, `@react-native-community/datetimepicker`, TypeScript

**Skills:** `@expo-best-practices` (image picker, permissions), `@mobile-developer` (form patterns, DateTimePicker), `@uncodixfy` (invoke before writing any JSX/styling — no generic AI aesthetics), `@code-quality` (TypeScript strictness)

```ts
import { Timestamp, addDoc, collection, updateDoc, doc, serverTimestamp } from 'firebase/firestore';
import { ref, uploadBytes, getDownloadURL } from 'firebase/storage';
import { db, storage } from '@/lib/firebase';
import { encodeGeohash } from '@/lib/geohash';

interface ListingInput {
  userId: string;
  title: string;
  description: string;
  images: string[];       // already-uploaded Storage URLs
  address: string;
  lat: number;
  lng: number;
  startDate: Timestamp;
  endDate: Timestamp | null;
  startTime: string;
  endTime: string;
}

export function buildListingPayload(input: ListingInput) {
  return {
    ...input,
    geohash: encodeGeohash(input.lat, input.lng),
    expiresAt: input.endDate ?? input.startDate,
    active: true,
    createdAt: serverTimestamp(),
  };
}

export async function uploadListingImages(uris: string[], userId: string): Promise<string[]> {
  const urls: string[] = [];
  for (const uri of uris) {
    const response = await fetch(uri);
    const blob = await response.blob();
    const storageRef = ref(storage, `listings/${userId}/${Date.now()}_${Math.random()}`);
    await uploadBytes(storageRef, blob);
    const url = await getDownloadURL(storageRef);
    urls.push(url);
  }
  return urls;
}

export async function createListing(input: ListingInput): Promise<string> {
  const payload = buildListingPayload(input);
  const docRef = await addDoc(collection(db, 'listings'), payload);
  return docRef.id;
}

export async function updateListing(listingId: string, input: Partial<ListingInput> & { endDate?: Timestamp | null }): Promise<void> {
  const updates: Record<string, any> = { ...input };
  if ('startDate' in input || 'endDate' in input) {
    updates.expiresAt = input.endDate ?? input.startDate;
  }
  if ('lat' in input && 'lng' in input) {
    updates.geohash = encodeGeohash(input.lat!, input.lng!);
  }
  await updateDoc(doc(db, 'listings', listingId), updates);
}
```

```ts
import { Timestamp } from 'firebase/firestore';

interface FormValues {
  title: string;
  description: string;
  images: string[];
  address: string;
  lat?: number;
  lng?: number;
  startDate: Timestamp;
  endDate: Timestamp | null;
  startTime: string;
  endTime: string;
}

export function validateListingForm(values: FormValues): Record<string, string> {
  const errors: Record<string, string> = {};
  if (!values.title.trim()) errors.title = 'Title is required.';
  if (values.title.length > 100) errors.title = 'Title must be under 100 characters.';
  if (!values.description.trim()) errors.description = 'Description is required.';
  if (values.description.length > 500) errors.description = 'Description must be under 500 characters.';
  if (values.images.length === 0) errors.images = 'At least one image is required.';
  if (!values.address || !values.lat || !values.lng) errors.address = 'Select an address from the suggestions.';
  if (values.startDate.toMillis() < Date.now()) errors.startDate = 'Start date must not be in the past.';
  if (values.endDate && values.endDate.toMillis() <= values.startDate.toMillis()) {
    errors.endDate = 'Second day must be after the first day.';
  }
  const [startH, startM] = values.startTime.split(':').map(Number);
  const [endH, endM] = values.endTime.split(':').map(Number);
  if (endH * 60 + endM <= startH * 60 + startM) errors.endTime = 'End time must be after start time.';
  return errors;
}
```

```tsx
import { useState } from 'react';
import {
  View, Text, TextInput, TouchableOpacity, ScrollView,
  Image, StyleSheet, ActivityIndicator, Alert, Platform
} from 'react-native';
import * as ImagePicker from 'expo-image-picker';
import DateTimePicker from '@react-native-community/datetimepicker';
import { GooglePlacesAutocomplete } from 'react-native-google-places-autocomplete';
import { Timestamp } from 'firebase/firestore';
import { useRouter, useLocalSearchParams } from 'expo-router';
import { auth } from '@/lib/firebase';
import { uploadListingImages, createListing, updateListing } from '@/lib/listings';
import { validateListingForm } from '@/lib/validation';
import Constants from 'expo-constants';

const PLACES_KEY = Constants.expoConfig?.extra?.googlePlacesApiKey ?? '';

export default function AddListingScreen() {
  const router = useRouter();
  const { listingId } = useLocalSearchParams<{ listingId?: string }>();
  const isEdit = !!listingId;

  const [title, setTitle] = useState('');
  const [description, setDescription] = useState('');
  const [images, setImages] = useState<string[]>([]);           // local URIs for new, URLs for existing
  const [existingImageUrls, setExistingImageUrls] = useState<string[]>([]);
  const [address, setAddress] = useState('');
  const [lat, setLat] = useState<number | null>(null);
  const [lng, setLng] = useState<number | null>(null);
  const [startDate, setStartDate] = useState<Date>(new Date(Date.now() + 86400000));
  const [endDate, setEndDate] = useState<Date | null>(null);
  const [startTime, setStartTime] = useState('09:00');
  const [endTime, setEndTime] = useState('14:00');
  const [showStartDate, setShowStartDate] = useState(false);
  const [showEndDate, setShowEndDate] = useState(false);
  const [showStartTime, setShowStartTime] = useState(false);
  const [showEndTime, setShowEndTime] = useState(false);

  // Helpers for time picker ↔ "HH:mm" string conversion
  const timeStringToDate = (t: string): Date => {
    const [h, m] = t.split(':').map(Number);
    const d = new Date(); d.setHours(h, m, 0, 0); return d;
  };
  const dateToTimeString = (d: Date): string =>
    `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;
  const [errors, setErrors] = useState<Record<string, string>>({});
  const [submitting, setSubmitting] = useState(false);

  const pickImage = async () => {
    if (images.length + existingImageUrls.length >= 3) return;
    const result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      quality: 0.8,
    });
    if (!result.canceled) setImages((prev) => [...prev, result.assets[0].uri]);
  };

  const removeImage = (index: number, isExisting: boolean) => {
    if (isExisting) setExistingImageUrls((prev) => prev.filter((_, i) => i !== index));
    else setImages((prev) => prev.filter((_, i) => i !== index));
  };

  const handleSubmit = async () => {
    if (!auth.currentUser) return;
    const allImages = [...existingImageUrls, ...images];
    const formValues = {
      title, description, images: allImages,
      address, lat: lat ?? undefined, lng: lng ?? undefined,
      startDate: Timestamp.fromDate(startDate),
      endDate: endDate ? Timestamp.fromDate(endDate) : null,
      startTime, endTime,
    };
    const validationErrors = validateListingForm(formValues as any);
    if (Object.keys(validationErrors).length > 0) { setErrors(validationErrors); return; }

    setSubmitting(true);
    setErrors({});
    try {
      // Upload only new local images
      let uploadedUrls: string[] = [];
      if (images.length > 0) {
        uploadedUrls = await uploadListingImages(images, auth.currentUser.uid);
      }
      const finalImages = [...existingImageUrls, ...uploadedUrls];
      const payload = {
        userId: auth.currentUser.uid,
        title, description,
        images: finalImages,
        address, lat: lat!, lng: lng!,
        startDate: Timestamp.fromDate(startDate),
        endDate: endDate ? Timestamp.fromDate(endDate) : null,
        startTime, endTime,
      };
      if (isEdit && listingId) {
        await updateListing(listingId, payload);
      } else {
        await createListing(payload);
      }
      router.replace('/(tabs)');
    } catch (e) {
      Alert.alert('Error', 'Failed to save listing. Please try again.');
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <ScrollView style={styles.container} keyboardShouldPersistTaps="handled">
      <Text style={styles.heading}>{isEdit ? 'Edit Listing' : 'Add Garage Sale'}</Text>

      {/* Images */}
      <View style={styles.imageRow}>
        {existingImageUrls.map((uri, i) => (
          <View key={`e-${i}`} style={styles.imageThumb}>
            <Image source={{ uri }} style={styles.thumbImg} />
            <TouchableOpacity style={styles.removeBtn} onPress={() => removeImage(i, true)}>
              <Text style={styles.removeBtnText}>✕</Text>
            </TouchableOpacity>
          </View>
        ))}
        {images.map((uri, i) => (
          <View key={`n-${i}`} style={styles.imageThumb}>
            <Image source={{ uri }} style={styles.thumbImg} />
            <TouchableOpacity style={styles.removeBtn} onPress={() => removeImage(i, false)}>
              <Text style={styles.removeBtnText}>✕</Text>
            </TouchableOpacity>
          </View>
        ))}
        {images.length + existingImageUrls.length < 3 && (
          <TouchableOpacity style={styles.addImageBtn} onPress={pickImage}>
            <Text style={styles.addImageText}>+ Photo</Text>
          </TouchableOpacity>
        )}
      </View>
      {errors.images && <Text style={styles.error}>{errors.images}</Text>}

      {/* Title */}
      <TextInput style={styles.input} placeholder="Title" value={title} onChangeText={setTitle} />
      {errors.title && <Text style={styles.error}>{errors.title}</Text>}

      {/* Description */}
      <TextInput style={[styles.input, styles.multiline]} placeholder="Description" value={description} onChangeText={setDescription} multiline numberOfLines={4} />
      {errors.description && <Text style={styles.error}>{errors.description}</Text>}

      {/* Address */}
      <GooglePlacesAutocomplete
        placeholder="Search address"
        onPress={(data, details) => {
          if (!details) return;
          setAddress(data.description);
          setLat(details.geometry.location.lat);
          setLng(details.geometry.location.lng);
        }}
        query={{ key: PLACES_KEY, language: 'en' }}
        fetchDetails
        styles={{ container: { marginBottom: 12 }, textInput: styles.input }}
      />
      {errors.address && <Text style={styles.error}>{errors.address}</Text>}

      {/* Start date — read-only in edit mode if sale has already started */}
      {isEdit && startDate < new Date() ? (
        <View style={[styles.input, { backgroundColor: '#f3f4f6' }]}>
          <Text style={{ color: '#9ca3af' }}>{`Start: ${startDate.toLocaleDateString('en-AU')} (cannot change — sale already started)`}</Text>
        </View>
      ) : (
        <TouchableOpacity style={styles.input} onPress={() => setShowStartDate(true)}>
          <Text>{`Start: ${startDate.toLocaleDateString('en-AU')}`}</Text>
        </TouchableOpacity>
      )}
      {showStartDate && (
        <DateTimePicker value={startDate} mode="date" minimumDate={new Date()} onChange={(_, d) => { setShowStartDate(false); if (d) setStartDate(d); }} />
      )}
      {errors.startDate && <Text style={styles.error}>{errors.startDate}</Text>}

      {/* Optional second day */}
      <TouchableOpacity style={styles.secondDayToggle} onPress={() => setEndDate(endDate ? null : new Date(startDate.getTime() + 86400000))}>
        <Text style={styles.secondDayText}>{endDate ? '✓ Two-day sale' : '+ Add second day'}</Text>
      </TouchableOpacity>
      {endDate && (
        <>
          <TouchableOpacity style={styles.input} onPress={() => setShowEndDate(true)}>
            <Text>{`End: ${endDate.toLocaleDateString('en-AU')}`}</Text>
          </TouchableOpacity>
          {showEndDate && (
            <DateTimePicker value={endDate} mode="date" minimumDate={new Date(startDate.getTime() + 86400000)} onChange={(_, d) => { setShowEndDate(false); if (d) setEndDate(d); }} />
          )}
        </>
      )}
      {errors.endDate && <Text style={styles.error}>{errors.endDate}</Text>}

      {/* Times — use DateTimePicker in time mode for reliable parsing */}
      {/* Store as "HH:mm" string derived from a Date object for picker */}
      <View style={styles.timeRow}>
        <TouchableOpacity style={[styles.input, { flex: 1 }]} onPress={() => setShowStartTime(true)}>
          <Text>Start: {startTime}</Text>
        </TouchableOpacity>
        <TouchableOpacity style={[styles.input, { flex: 1, marginLeft: 8 }]} onPress={() => setShowEndTime(true)}>
          <Text>End: {endTime}</Text>
        </TouchableOpacity>
      </View>
      {showStartTime && (
        <DateTimePicker
          value={timeStringToDate(startTime)}
          mode="time"
          is24Hour
          onChange={(_, d) => { setShowStartTime(false); if (d) setStartTime(dateToTimeString(d)); }}
        />
      )}
      {showEndTime && (
        <DateTimePicker
          value={timeStringToDate(endTime)}
          mode="time"
          is24Hour
          onChange={(_, d) => { setShowEndTime(false); if (d) setEndTime(dateToTimeString(d)); }}
        />
      )}
      {errors.endTime && <Text style={styles.error}>{errors.endTime}</Text>}

      <TouchableOpacity style={styles.submitBtn} onPress={handleSubmit} disabled={submitting}>
        {submitting ? <ActivityIndicator color="#fff" /> : <Text style={styles.submitBtnText}>{isEdit ? 'Save Changes' : 'Add Listing'}</Text>}
      </TouchableOpacity>
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#fff', padding: 16 },
  heading: { fontSize: 22, fontWeight: '700', marginBottom: 16 },
  imageRow: { flexDirection: 'row', gap: 8, marginBottom: 4 },
  imageThumb: { position: 'relative' },
  thumbImg: { width: 80, height: 80, borderRadius: 8 },
  removeBtn: { position: 'absolute', top: -6, right: -6, backgroundColor: '#ef4444', borderRadius: 10, width: 20, height: 20, alignItems: 'center', justifyContent: 'center' },
  removeBtnText: { color: '#fff', fontSize: 11, fontWeight: '700' },
  addImageBtn: { width: 80, height: 80, borderRadius: 8, borderWidth: 1, borderStyle: 'dashed', borderColor: '#9ca3af', alignItems: 'center', justifyContent: 'center' },
  addImageText: { color: '#6b7280', fontSize: 13 },
  input: { borderWidth: 1, borderColor: '#e5e7eb', borderRadius: 8, padding: 12, marginBottom: 8, backgroundColor: '#fff' },
  multiline: { height: 100, textAlignVertical: 'top' },
  timeRow: { flexDirection: 'row' },
  secondDayToggle: { marginBottom: 8 },
  secondDayText: { color: '#2563eb', fontSize: 14 },
  error: { color: '#dc2626', fontSize: 12, marginBottom: 8, marginTop: -4 },
  submitBtn: { backgroundColor: '#2563eb', padding: 16, borderRadius: 8, alignItems: 'center', marginTop: 8, marginBottom: 32 },
  submitBtnText: { color: '#fff', fontWeight: '600', fontSize: 16 },
});
```
